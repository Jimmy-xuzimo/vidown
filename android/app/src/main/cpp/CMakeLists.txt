cmake_minimum_required(VERSION 3.22)
project(vidown_media LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../..")
get_filename_component(ROOT "${ROOT}" ABSOLUTE)
set(PREFIX "${ROOT}/native-build/prefix/${ANDROID_ABI}")
if(NOT EXISTS "${PREFIX}/lib/libavformat.so")
  message(FATAL_ERROR "Run android/tools/prepare-sources.py and build-native.sh before building the APK")
endif()
foreach(lib avformat avcodec avutil swresample)
  add_library(${lib} SHARED IMPORTED)
  set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${PREFIX}/lib/lib${lib}.so")
endforeach()
set(QJS "${ROOT}/third-party/quickjs")
add_library(vidown_qjs STATIC "${QJS}/quickjs.c" "${QJS}/dtoa.c" "${QJS}/libregexp.c" "${QJS}/libunicode.c")
target_compile_definitions(vidown_qjs PRIVATE _GNU_SOURCE QUICKJS_NG_BUILD)
target_include_directories(vidown_qjs PUBLIC "${QJS}")
set_target_properties(vidown_qjs PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(vidown_media SHARED media.cpp)
target_include_directories(vidown_media PRIVATE "${PREFIX}/include")
target_link_libraries(vidown_media avformat avcodec avutil swresample vidown_qjs android log m)
target_link_options(vidown_media PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
